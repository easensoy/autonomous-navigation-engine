# expected outcome is the engine's BLOCKED state and a controlled halt, not a completed mission.

event x_ge -3.0 spawn_obstacle pallet_c 1.0 0.0 1.0 1.2 1.0
accept outcome == BLOCKED
accept collisions == 0
accept interventions == 0
accept operator_blockage_reports == 2
accept sensor_blockage_reports >= 1
accept stop_time_after_terminal_s <= 1.0
accept drift_after_stop_m <= 0.05
