# must notice within its 0.25 s feedback timeout, command zero velocity, hold, and resume once
# feedback returns (inside its 4.0 s recovery window) with the pose still on the route.

accept goal_reached == true
accept final_position_error_m <= 0.10
accept collisions == 0
accept interventions == 0
accept feedback_holds == 1
accept feedback_recoveries == 1
accept hold_detect_s <= 0.35
accept blind_travel_m <= 0.35
