# ane-layout 1
#
# Indoor warehouse for the ROS 2 / Gazebo integration. This file is the single source of the
# warehouse geometry: scripts/generate_world.py turns it into the Gazebo world (worlds/warehouse.sdf)
# and into the planning roadmap the engine loads (maps/warehouse_roadmap.txt).
#
# Every length is in metres in the map frame. The map frame is the Gazebo world frame: x east, y north,
# origin at the centre of the hall.

name ane_warehouse

# Interior floor of the hall (min_x min_y max_x max_y); walls stand outside it.
hall -8.0 -5.0 8.0 5.0
# Wall thickness and height.
wall 0.2 1.5

# Static obstacles: kind min_x min_y max_x max_y height label.
# The two middle rack rows separate three east-west aisles (centre lines y = 3, 0, -3, each 1.8 m
# wide); the wall rack rows bound the outer aisles.
rack  -3.5  0.9  3.5  2.1  2.0  rack_north
rack  -3.5 -2.1  3.5 -0.9  2.0  rack_south
rack  -3.5  3.9  3.5  4.9  2.0  rack_wall_north
rack  -3.5 -4.9  3.5 -3.9  2.0  rack_wall_south
table -8.0 -0.6 -7.2  0.6  0.8  dispatch_table
table  7.2 -0.6  8.0  0.6  0.8  delivery_conveyor

# Roadmap lanes: straight centre lines the robot may drive (x0 y0 x1 y1). Lane end points and lane
# crossings become shared graph nodes; every lane is then divided into pieces of at most node_spacing.
lane west_cross    -5.0 -3.0  -5.0  3.0
lane east_cross     5.0 -3.0   5.0  3.0
lane aisle_north   -5.0  3.0   5.0  3.0
lane aisle_middle  -5.0  0.0   5.0  0.0
lane aisle_south   -5.0 -3.0   5.0 -3.0
lane dispatch_spur -6.5  0.0  -5.0  0.0
lane delivery_spur  5.0  0.0   6.5  0.0
node_spacing 0.5

# Named stations: the robot is dispatched from one and delivers to the other. Each must lie on a node.
station dispatch -6.5 0.0
station delivery  6.5 0.0

# Cost zones (min_x min_y max_x max_y multiplier label): an edge whose midpoint lies in the zone costs
# its length times the multiplier. The north aisle crosses a picking zone where people work, so the
# south aisle is the cheaper detour although both detours have the same length.
zone -2.0 2.1 2.0 3.9 1.5 picking_zone

# Graph frame: the engine stores node coordinates in decimetres from the hall's south-west corner.
# map = R(yaw) * (scale * graph) + origin
graph_frame 0.1 0.0 -8.0 -5.0
